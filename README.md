## استفاده

```php
use Tavaan\Engine\Module\ModuleRegistry;
use Tavaan\Engine\Module\Module;
use Tavaan\Engine\Module\ManifestParser;
use Tavaan\Engine\Module\TopologicalSorter;

// کشف خودکار ماژول‌ها
$registry = new ModuleRegistry();
$registry->discover('modules/');  // تمام ماژول‌ها با module.json

// دسترسی
$registry->has('tavaan.shop');
$registry->get('tavaan.shop');    // throws if not found
$registry->all();                 // list<Module>
$registry->enabled();             // list<Module> enabled only

// Enable/Disable
$registry->enable('tavaan.shop');
$registry->disable('tavaan.shop');

// مرتب‌سازی topological (وابستگی‌ها اول)
$registry->sorted();              // list<Module> in dependency order

// وابستگی‌ها
$registry->dependencies('tavaan.shop');          // ['tavaan.core-user', ...]
$registry->dependentsOf('tavaan.core-user');     // ['tavaan.shop', 'tavaan.media', ...]

// دستی (بدون discover)
$parser = new ManifestParser();
$module = $parser->parse('/path/to/module.json');
$registry->register($module);

// TopologicalSorter standalone
$sorter = new TopologicalSorter();
$sorted = $sorter->sort($modulesMap);  // returns list<string> of IDs
```
